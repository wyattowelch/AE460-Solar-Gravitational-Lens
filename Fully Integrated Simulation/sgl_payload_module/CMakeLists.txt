cmake_minimum_required(VERSION 3.16)
project(sgl_payload_module LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(sgl_payload_module src/payload_module.cpp)
target_include_directories(sgl_payload_module PUBLIC include)
add_executable(payload_standalone_demo examples/standalone_demo.cpp)
target_link_libraries(payload_standalone_demo PRIVATE sgl_payload_module)
enable_testing()
add_executable(test_payload tests/test_payload.cpp)
target_link_libraries(test_payload PRIVATE sgl_payload_module)
add_test(NAME test_payload COMMAND test_payload)
