cmake_minimum_required(VERSION 3.16)
project(sgl_thermal_module LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(sgl_thermal_module src/thermal_module.cpp)
target_include_directories(sgl_thermal_module PUBLIC include)
add_executable(thermal_standalone_demo examples/standalone_demo.cpp)
target_link_libraries(thermal_standalone_demo PRIVATE sgl_thermal_module)
enable_testing()
add_executable(test_thermal tests/test_thermal.cpp)
target_link_libraries(test_thermal PRIVATE sgl_thermal_module)
add_test(NAME test_thermal COMMAND test_thermal)
