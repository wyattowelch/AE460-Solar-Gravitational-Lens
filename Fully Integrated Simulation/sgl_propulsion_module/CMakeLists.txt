cmake_minimum_required(VERSION 3.16)
project(sgl_propulsion_module LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(sgl_propulsion_module src/propulsion_module.cpp)
target_include_directories(sgl_propulsion_module PUBLIC include)
add_executable(propulsion_standalone_demo examples/standalone_demo.cpp)
target_link_libraries(propulsion_standalone_demo PRIVATE sgl_propulsion_module)
enable_testing()
add_executable(test_propulsion tests/test_propulsion.cpp)
target_link_libraries(test_propulsion PRIVATE sgl_propulsion_module)
add_test(NAME test_propulsion COMMAND test_propulsion)
